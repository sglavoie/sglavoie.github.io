{{- /* Markdown images, sized when the file is local. WebP versions are added afterwards, by partials/responsive-images.html. */ -}} {{- $destination := .Destination | safeURL -}} {{- $alt := .PlainText | default (.Text | plainify | htmlUnescape) | strings.TrimSpace -}} {{- $title := .Title -}} {{- $cleanDestination := .Destination | replaceRE "[?#].*$" "" -}} {{- $isRemote := or (hasPrefix .Destination "http://") (hasPrefix .Destination "https://") -}} {{- $width := 0 -}} {{- $height := 0 -}} {{- if not $isRemote -}} {{- $relativePath := strings.TrimPrefix "/" $cleanDestination -}} {{- if hasPrefix $cleanDestination "/" -}} {{- $assetPath := path.Join "static" $relativePath -}} {{- $extension := lower (path.Ext $cleanDestination) -}} {{- if and (fileExists $assetPath) (in (slice ".png" ".jpg" ".jpeg" ".gif" ".webp") $extension) -}} {{- $imageConfig := imageConfig $assetPath -}} {{- $width = $imageConfig.Width -}} {{- $height = $imageConfig.Height -}} {{- end -}} {{- end -}} {{- end -}}
